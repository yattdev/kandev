class Unqualified(Exception):
    pass


def require(value, reason):
    if not value:
        raise Unqualified(reason)


def unescape(value):
    return re.sub(r'\\([0-7]{3})', lambda m: chr(int(m[1], 8)), value)


def descriptor_mount(fd, root):
    info = Path(f'/proc/self/fdinfo/{fd}').read_text()
    match = re.search(r'^mnt_id:\s+(\d+)$', info, re.M)
    require(match, 'MOUNT_ID_UNAVAILABLE')
    mount_id = match[1]
    rows = [line.split(' - ', 1)[0].split()
            for line in Path('/proc/self/mountinfo').read_text().splitlines()]
    found = [row for row in rows if row[0] == mount_id]
    require(len(found) == 1, 'MOUNT_ID_UNKNOWN')
    row = found[0]
    require(unescape(row[4]) == str(root), 'ROOT_NOT_EXACT_MOUNT')
    require('rw' in row[5].split(','), 'ROOT_MOUNT_READ_ONLY')
    require(not os.fstatvfs(fd).f_flag & os.ST_RDONLY, 'ROOT_FS_READ_ONLY')
    return mount_id


def qualify(git='git'):
    # Never alter caller identity, remove a context override, or read a token.
    require(os.environ.get('KANDEV_AGENT_GUARD_ACTIVE') == '1' and
            os.environ.get('KANDEV_AGENT_GUARD_SCOPE') == 'project', 'GUARD_CONTEXT_UNKNOWN')
    hint = os.environ.get('KANDEV_AGENT_DOCKER_ROOT')
    require(hint, 'SCOPE_HINT_UNAVAILABLE')
    root = Path(hint).resolve(strict=True)
    require(root.is_absolute() and str(root) == hint, 'ROOT_HINT_NOT_PHYSICAL')
    cwd = Path.cwd().resolve(strict=True)
    require(cwd == root or root in cwd.parents, 'CWD_OUTSIDE_SCOPE')
    # Accept only the two non-discovery Git settings exported by this guard.
    git_env = {key: value for key, value in os.environ.items() if key.startswith('GIT_')}
    guard_git = {'GIT_CONFIG_COUNT': '2', 'GIT_CONFIG_KEY_0': 'gc.auto',
                 'GIT_CONFIG_VALUE_0': '0', 'GIT_CONFIG_KEY_1': 'maintenance.auto',
                 'GIT_CONFIG_VALUE_1': 'false'}
    require(git_env in ({}, guard_git), 'GIT_CONTEXT_UNQUALIFIED')
    env = dict(os.environ, LC_ALL='C')

    def read_git(directory, flag):
        return subprocess.run([git, '--no-pager', '--no-optional-locks', '-C',
                               str(directory), 'rev-parse', flag],
                              env=env, text=True, capture_output=True, timeout=10)

    def git_relation():
        current = read_git(cwd, '--show-toplevel')
        require(current.returncode == 0, 'CHECKOUT_CONTEXT_UNKNOWN')
        checkout = Path(current.stdout.strip()).resolve(strict=True)
        require(root in checkout.parents, 'ROOT_NOT_OUTSIDE_CHECKOUT')
        result = read_git(root, '--is-inside-work-tree')
        # No generic exit-128 -> absence conversion. Mount-boundary absence is
        # recognized explicitly; every permission/config/transport error stops.
        lines = result.stderr.splitlines()
        known_absence = (lines == ['fatal: not a git repository (or any of the parent directories): .git'] or
                         (len(lines) == 2 and
                          lines[0].startswith('fatal: not a git repository (or any parent up to mount point ') and
                          lines[0].endswith(')') and
                          lines[1] == 'Stopping at filesystem boundary (GIT_DISCOVERY_ACROSS_FILESYSTEM not set).'))
        require(result.returncode == 128 and not result.stdout and known_absence,
                'ROOT_GIT_CONTEXT_NOT_ABSENT')
        return checkout

    fd = os.open(root, os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC | os.O_NOFOLLOW)
    try:
        identity = os.fstat(fd)
        mount_id = descriptor_mount(fd, root)
        checkout = git_relation()
        require(hasattr(os, 'O_TMPFILE'), 'ANONYMOUS_PROBE_UNSUPPORTED')
        try:
            anonymous = os.open('.', os.O_TMPFILE | os.O_RDWR | os.O_CLOEXEC,
                                0o600, dir_fd=fd)
        except OSError as error:
            raise Unqualified('ANONYMOUS_PROBE_UNQUALIFIED:' + errno.errorcode.get(error.errno, 'UNKNOWN')) from error
        try:
            probe = os.fstat(anonymous)
            require(probe.st_nlink == 0 and probe.st_size == 0, 'ANONYMOUS_PROBE_INVALID')
        finally:
            os.close(anonymous)
        after = root.stat()
        require((after.st_dev, after.st_ino) == (identity.st_dev, identity.st_ino), 'ROOT_CHANGED')
        require(descriptor_mount(fd, root) == mount_id and git_relation() == checkout,
                'BOUNDARY_CHANGED')
        return {'result': 'QUALIFIED_LOCAL_OBSERVATION', 'root': str(root),
                'checkout': str(checkout), 'root_dev': identity.st_dev,
                'root_ino': identity.st_ino, 'mount_id': mount_id,
                'outside_git': True, 'zero_byte_anonymous_write': True,
                'persisted_paths': 0, 'authenticated_task_session_grant': False}
    finally:
        os.close(fd)


def fixture():
    results = []
    with tempfile.TemporaryDirectory(prefix='ec7-private-') as temporary:
        base = Path(temporary)
        task = base / 'task with space'
        repo = task / 'checkout'
        repo.mkdir(parents=True)
        env = {'PATH': '/usr/bin:/bin', 'HOME': str(base), 'LC_ALL': 'C'}
        subprocess.run(['/usr/bin/git', 'init', '-q', str(repo)], env=env, check=True)
        (repo / '.gitignore').write_text('ignored/\n')
        (repo / 'ignored').mkdir()
        (base / 'root-alias').symlink_to(task, target_is_directory=True)
        # Exercise real production routing; only replace the broker with an
        # inert failure tripwire in the private copy (same existing test seam).
        marker = task / 'BROKER_WAS_CALLED'
        broker = base / 'broker'
        broker.write_text('#!/bin/sh\ntouch "' + str(marker) + '"\nexit 78\n')
        broker.chmod(0o700)
        wrapper = base / 'git'
        wrapper.write_text(GIT_WRAPPER.read_text().replace('/usr/local/bin/kandev-agent-docker-client', str(broker)))
        wrapper.chmod(0o700)
        fake = base / 'git-error'
        fake.write_text('#!/bin/sh\ncase "$*" in\n*--show-toplevel*) printf "%s\\n" "' + str(repo) + '";;\n*) echo "fatal: synthetic permission denial" >&2; exit 128;;\nesac\n')
        fake.chmod(0o700)
        (repo / '.git').chmod(0o555)
        before = {str(p.relative_to(task)): hashlib.sha256(p.read_bytes()).hexdigest()
                  for p in task.rglob('*') if p.is_file()}
        common = ['bwrap', '--die-with-parent', '--unshare-net', '--ro-bind', '/', '/',
                  '--dev', '/dev', '--proc', '/proc', '--bind', str(task), str(task), '--chdir', str(repo)]
        env.update(KANDEV_AGENT_GUARD_ACTIVE='1', KANDEV_AGENT_GUARD_SCOPE='project',
                   KANDEV_AGENT_DOCKER_ROOT=str(task), KANDEV_AGENT_DOCKER_TOKEN='private-synthetic-value',
                   GIT_CONFIG_COUNT='2', GIT_CONFIG_KEY_0='gc.auto', GIT_CONFIG_VALUE_0='0',
                   GIT_CONFIG_KEY_1='maintenance.auto', GIT_CONFIG_VALUE_1='false')
        cases = [
            ('exact-rw-root', {}, [], str(wrapper), None),
            ('filesystem-boundary-root', {}, ['--tmpfs', str(base), '--bind', str(task), str(task), '--ro-bind', str(wrapper), str(wrapper), '--ro-bind', str(broker), str(broker)], str(wrapper), None),
            ('missing-hint', {'KANDEV_AGENT_DOCKER_ROOT': ''}, [], str(wrapper), 'SCOPE_HINT_UNAVAILABLE'),
            ('hint-is-checkout', {'KANDEV_AGENT_DOCKER_ROOT': str(repo)}, ['--bind', str(repo), str(repo)], str(wrapper), 'ROOT_NOT_OUTSIDE_CHECKOUT'),
            ('ignored-inside-checkout', {'KANDEV_AGENT_DOCKER_ROOT': str(repo / 'ignored')}, ['--bind', str(repo / 'ignored'), str(repo / 'ignored'), '--chdir', str(repo / 'ignored')], str(wrapper), 'ROOT_NOT_OUTSIDE_CHECKOUT'),
            ('symlink-hint', {'KANDEV_AGENT_DOCKER_ROOT': str(base / 'root-alias')}, [], str(wrapper), 'ROOT_HINT_NOT_PHYSICAL'),
            ('hint-outside-cwd', {'KANDEV_AGENT_DOCKER_ROOT': str(base / 'other')}, [], str(wrapper), 'CWD_OUTSIDE_SCOPE'),
            ('ancestor-is-not-mount', {'KANDEV_AGENT_DOCKER_ROOT': str(base)}, [], str(wrapper), 'ROOT_NOT_EXACT_MOUNT'),
            ('root-mounted-readonly', {}, ['--ro-bind', str(task), str(task)], str(wrapper), 'ROOT_MOUNT_READ_ONLY'),
            ('git-context-override', {'GIT_DIR': str(repo / '.git')}, [], str(wrapper), 'GIT_CONTEXT_UNQUALIFIED'),
            ('generic-git-error', {}, [], str(fake), 'ROOT_GIT_CONTEXT_NOT_ABSENT'),
            ('root-is-itself-a-repository', {}, [], str(wrapper), 'ROOT_GIT_CONTEXT_NOT_ABSENT'),
            ('anonymous-unsupported', {'EC7_FIXTURE_TMPFILE_ERROR': str(errno.EOPNOTSUPP)}, [], str(wrapper), 'ANONYMOUS_PROBE_UNQUALIFIED:ENOTSUP'),
            ('anonymous-permission-denied', {'EC7_FIXTURE_TMPFILE_ERROR': str(errno.EACCES)}, [], str(wrapper), 'ANONYMOUS_PROBE_UNQUALIFIED:EACCES'),
        ]
        (base / 'other').mkdir()
        try:
            for name, changes, extra, git, expected in cases:
                if name == 'root-is-itself-a-repository':
                    subprocess.run(['/usr/bin/git', 'init', '-q', str(task)], env={'PATH': '/usr/bin:/bin', 'HOME': str(base)}, check=True)
                try:
                    result = subprocess.run(common + extra + ['/usr/bin/python3', __file__, '--probe', git],
                                            env=dict(env, **changes), text=True, capture_output=True, timeout=20)
                finally:
                    if name == 'root-is-itself-a-repository':
                        shutil.rmtree(task / '.git')
                receipt = json.loads(result.stdout)
                assert receipt.get('error') == expected, (name, result.returncode, result.stdout, result.stderr)
                assert result.returncode == (0 if expected is None else 78), (name, result)
                results.append({'case': name, 'result': receipt.get('error', receipt.get('result'))})
                reference = os.environ.get('EC7_REFERENCE_FILE')
                if reference and expected is None:
                    exact = subprocess.run(common + extra + ['/usr/bin/python3', reference],
                                           env=dict(env, PATH=str(base) + ':/usr/bin:/bin'),
                                           text=True, capture_output=True, timeout=20)
                    exact_receipt = json.loads(exact.stdout)
                    assert exact.returncode == 0 and exact_receipt['outside_git'], (name, exact.stdout, exact.stderr)
                    results.append({'case': name + '-exact-reference', 'result': exact_receipt['result']})
            assert not marker.exists(), 'native readonly Git attempted broker lookup'
            after = {str(p.relative_to(task)): hashlib.sha256(p.read_bytes()).hexdigest()
                     for p in task.rglob('*') if p.is_file()}
            assert before == after, 'probe persisted files or changed repository content'
        finally:
            (repo / '.git').chmod(0o755)
    print(json.dumps({'cases': results, 'passed': len(results), 'target_operations': 0,
                      'broker_calls': 0, 'named_file_and_content_changes': 0}, indent=2))


if __name__ == '__main__':
    if len(sys.argv) > 1 and sys.argv[1] == '--probe':
        try:
            # These two injections exist only in the private fixture, never in
            # a proposed owner operation. No real engine, token or DB is used.
            number = os.environ.get('EC7_FIXTURE_TMPFILE_ERROR')
            real_open = os.open
            def fixture_open(path, flags, *args, **kwargs):
                if number and flags & getattr(os, 'O_TMPFILE', 0) == getattr(os, 'O_TMPFILE', -1):
                    raise OSError(int(number), 'private fixture')
                return real_open(path, flags, *args, **kwargs)
            with patch('os.open', fixture_open):
                print(json.dumps(qualify(sys.argv[2])))
        except (Unqualified, OSError, subprocess.SubprocessError) as error:
            print(json.dumps({'error': str(error)}))
            sys.exit(78)
    else:
        fixture()
