NEW_ID = 'sha256:' + 'a'*64
OLD_ID = 'sha256:' + 'b'*64


class FreshBuildTests(unittest.TestCase):
    def setUp(self):
        temp = tempfile.TemporaryDirectory(prefix='fresh-bootstrap-')
        self.addCleanup(temp.cleanup)
        self.base = Path(temp.name).resolve()
        self.root = self.base/'task'
        self.private = self.root/'.kandev'
        self.context = self.private/'frozen'
        self.context.mkdir(parents=True)
        (self.context/'payload.txt').write_text('private synthetic current source\n')
        self.project = 'kd_fresh_bootstrap_fixture'
        self.tag = self.project+'_app:fresh-source-lock'
        self.base_image = 'docker.io/library/php:8.4-cli@sha256:' + 'c'*64
        self.composer_image = 'docker.io/library/composer:2@sha256:' + 'd'*64
        # Fake digests qualify admission only; not valid/resolved public image pins.
        self.dockerfile = (self.context/'Dockerfile')
        self.dockerfile.write_text('ARG PHP_BASE\nARG COMPOSER_BASE\n'
            'FROM ${COMPOSER_BASE} AS composer_tool\nFROM ${PHP_BASE}\n'
            'COPY --from=composer_tool /usr/bin/composer /usr/local/bin/composer\n'
            'WORKDIR /opt/current\nCOPY . .\n'
            'RUN php -r "exit(PHP_MAJOR_VERSION === 8 && PHP_MINOR_VERSION === 4 ? 0 : 1);"\n'
            '# An owner-approved recipe may fetch locked dev dependencies here.\n')
        self.original = {'services': {'db': {'image': 'fixture.invalid/db:10.11'},
                                     'app': {'image': OLD_ID}}}
        self.build = {'context': str(self.context), 'dockerfile': 'Dockerfile', 'pull': True,
                      'args': {'PHP_BASE': self.base_image, 'COMPOSER_BASE': self.composer_image}}
        self.overlay = {'services': {'app': {'image': self.tag, 'build': self.build}}}
        self.base_file, self.overlay_file = self.private/'base.yml', self.private/'fresh.yml'
        self.base_file.write_text(json.dumps(self.original))
        self.overlay_file.write_text(json.dumps(self.overlay))
        self.key = b'private-fixture-key-not-live'
        self.request = {'task_root': str(self.root), 'cwd': str(self.root),
            'token': broker.expected_token(self.key, self.root),
            'compose_environment': {'COMPOSE_PROJECT_NAME': self.project},
            'args': ['compose', '--project-directory', str(self.root), '-f', str(self.base_file),
                     '-f', str(self.overlay_file), 'build', '--pull', '--no-cache', 'app']}
        self.renders, self.replays, self.image_queries = [], [], []
        self.engine_exit = 0
        self.image_exists = False
        original_tempfile = tempfile.NamedTemporaryFile
        def private_tempfile(*args, **kwargs):
            self.assertEqual(kwargs.pop('dir'), '/run/kandev-agent-docker')
            return original_tempfile(*args, **kwargs, dir=self.base)
        for name, value in [('MODEL_DIR', self.base/'models'), ('COMPOSE_AUDIT_LOG', self.base/'audit.jsonl')]:
            patch = mock.patch.object(broker, name, value);patch.start();self.addCleanup(patch.stop)
        for patch in [mock.patch.object(broker, 'translate_to_host', side_effect=lambda p:p),
                      mock.patch.object(broker, 'run_command', side_effect=self.command),
                      mock.patch.object(broker.tempfile, 'NamedTemporaryFile', side_effect=private_tempfile)]:
            patch.start();self.addCleanup(patch.stop)

    def merged(self):
        result = copy.deepcopy(self.original)
        result['services']['app'].update(copy.deepcopy(self.overlay['services']['app']))
        return result

    def command(self, argv, cwd, **kwargs):
        if argv[0] == broker.GUARD_BIN:
            self.assertEqual(argv[1:3], ['--', broker.COMPOSE_PLUGIN])
            self.assertEqual(argv[-3:], ['config', '--format', 'json'])
            self.renders.append(argv)
            if config.installed:
                # Actual parser only; no daemon/socket/config/secret paths exist here.
                result = subprocess.run(argv[2:], cwd=cwd,
                    env={'PATH': '/usr/bin:/bin', 'HOME': str(self.base),
                         'DOCKER_CONFIG': str(self.base/'empty-config'),
                         **kwargs['environment_overrides']},
                    capture_output=True, text=True, timeout=15)
                return result.returncode, result.stdout, result.stderr
            return 0, json.dumps(self.merged()), ''
        self.assertEqual(argv[0], broker.DOCKER_BIN)
        if argv[1:3] == ['image', 'inspect']:
            self.image_queries.append(argv)
            self.assertNotEqual(argv[-1], OLD_ID, 'denied historical image reached engine')
            if not self.image_exists:
                return 1, '', 'No such image'
            if argv[-1] == self.tag:
                return 0, json.dumps(NEW_ID), ''
            self.assertEqual(argv[-1], NEW_ID)
            return 0, json.dumps({'Id': NEW_ID, 'RepoDigests': [], 'Os': 'linux',
                'Architecture': 'fixture', 'Created': 'fixture', 'Size': 1,
                'Config': {'WorkingDir': '/opt/current'}}), ''
        self.assertEqual(argv[1:4], ['compose', '--project-name', self.project])
        self.assertEqual(argv[-4:], ['build', '--pull', '--no-cache', 'app'])
        model = json.loads(Path(argv[argv.index('-f')+1]).read_text())
        # Registration happens before engine replay and is not success provenance.
        self.assertEqual(broker.load_scoped_model(self.root, self.root, self.project), model)
        self.assertEqual(model['services']['app']['image'], self.tag)
        self.assertEqual(model['services']['app']['build']['context'], str(self.context))
        self.assertEqual(model['services']['app']['build']['args']['PHP_BASE'], self.base_image)
        self.assertEqual(model['services']['app']['build']['args']['COMPOSER_BASE'], self.composer_image)
        self.assertEqual(model['services']['db']['image'], self.original['services']['db']['image'])
        self.replays.append(argv)
        self.image_exists = self.engine_exit == 0
        return self.engine_exit, 'MOCK_BUILD_ONLY\n', '' if self.engine_exit == 0 else 'synthetic build failure'

    def execute(self, **overrides):
        self.overlay_file.write_text(json.dumps(self.overlay))
        return broker.handle_request({**self.request, **overrides}, self.key)

    def inspect(self, identity):
        return broker.handle_request({**self.request, 'args':['image','inspect',identity]}, self.key)

    def test_fresh_build_has_no_historical_model_or_image_prerequisite(self):
        self.assertFalse(broker.MODEL_DIR.exists())
        self.assertEqual(self.execute()['returncode'], 0)
        self.assertEqual(len(self.replays), 1)
        expected = [self.tag] if hasattr(broker, 'successful_build_images') else []
        self.assertEqual([call[-1] for call in self.image_queries], expected)
        saved = broker.load_scoped_model(self.root, self.root, self.project)
        self.assertTrue(saved['services']['app']['build']['pull'])
        self.assertNotIn('network', saved['services']['app']['build'])
        self.assertEqual(saved['services']['app'].get('volumes', []), [])
        self.assertEqual(self.inspect(NEW_ID)['returncode'], 0)
        self.assertTrue(all(call[-1] in {self.tag,NEW_ID} for call in self.image_queries))

    def test_failed_build_preserves_registration_but_proves_no_image(self):
        self.engine_exit = 19
        self.assertEqual(self.execute()['returncode'], 19)
        self.assertTrue(broker.load_scoped_model(self.root,self.root,self.project)['services']['app']['build'])
        with self.assertRaisesRegex(broker.BrokerError, 'IMAGE_NOT_AUTHORIZED'):
            self.inspect(NEW_ID)
        self.assertFalse(any(call[-1] == NEW_ID for call in self.image_queries))

    def test_denied_old_id_is_not_admitted_by_new_build(self):
        self.execute()
        with self.assertRaisesRegex(broker.BrokerError, 'IMAGE_NOT_AUTHORIZED'):
            self.inspect(OLD_ID)
        self.assertFalse(any(call[-1] == OLD_ID for call in self.image_queries))

    def test_config_registers_model_without_build_or_output_proof(self):
        request = [*self.request['args'][:-4], 'config', '--format', 'json']
        # A config replay is a parser-only phase, intercepted separately below.
        original = self.command
        def config_only(argv,cwd,**kwargs):
            if argv[0] == broker.DOCKER_BIN:
                self.assertEqual(argv[-3:],['config','--format','json'])
                return 0, Path(argv[argv.index('-f')+1]).read_text(), ''
            return original(argv,cwd,**kwargs)
        with mock.patch.object(broker,'run_command',side_effect=config_only):
            self.assertEqual(self.execute(args=request)['returncode'],0)
        self.assertEqual(self.replays,[])
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):
            self.inspect(NEW_ID)

    def test_foreign_project_claim_cannot_be_bootstrapped(self):
        sibling=self.root/'sibling';sibling.mkdir()
        broker.claim_project(self.root,sibling,self.project)
        with self.assertRaisesRegex(broker.BrokerError,'different task clone'):
            self.execute()
        self.assertEqual(self.replays,[])

    def test_wrong_token_and_outside_cwd_deny_before_parser(self):
        for change in ({'token':'wrong'},{'cwd':str(self.base)}):
            with self.assertRaises(broker.BrokerError):self.execute(**change)
        self.assertEqual(self.renders,[])
        self.assertEqual(self.replays,[])

    def test_context_and_dockerfile_cannot_escape_task(self):
        for change in ({'context':str(self.base)}, {'dockerfile':str(self.base/'Dockerfile')}):
            self.overlay['services']['app']['build']={**self.build,**change}
            with self.assertRaises(broker.BrokerError):self.execute()
        self.assertEqual(self.replays,[])

    def test_reserved_build_grants_remain_denied(self):
        for field,value in [('ssh',['default']),('secrets',['secret']),('entitlements',['network.host']),
                            ('additional_contexts',{'foreign':'docker-image://sibling'}),
                            ('cache_from',['type=registry,ref=sibling']),('cache_to',['type=local,dest=/outside']),
                            ('tags',['shared:tag'])]:
            self.overlay['services']['app']['build']={**self.build,field:value}
            with self.assertRaises(broker.BrokerError):self.execute()
        self.assertEqual(self.replays,[])

    def test_cli_builder_secret_and_ssh_overrides_remain_denied(self):
        for flag in ['--builder=other','--secret=id=x','--ssh=default']:
            with self.assertRaises(broker.BrokerError):
                self.execute(args=[*self.request['args'][:-1],flag,'app'])
        self.assertEqual(self.renders,[])

    def test_other_task_project_claim_denies(self):
        other=self.base/'other';other.mkdir()
        broker.claim_project(other,other,self.project)
        with self.assertRaisesRegex(broker.BrokerError,'different task clone'):self.execute()
        self.assertEqual(self.replays,[])

    def test_installed_client_frame_uses_same_task_root_and_alias(self):
        if not config.installed:self.skipTest('installed wrapper/client qualified by installed mode')
        sock=self.base/'broker.sock'
        with broker.ThreadedUnixServer(str(sock),broker.BrokerHandler) as server:
            server.broker_key=self.key
            thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
            try:
                env={'PATH':'/usr/local/bin:/usr/bin:/bin','HOME':str(self.base),
                     'KANDEV_AGENT_GUARD_ACTIVE':'1','KANDEV_AGENT_GUARD_SCOPE':'project',
                     'KANDEV_AGENT_DOCKER_ROOT':str(self.root),
                     'KANDEV_AGENT_DOCKER_SOCKET':str(sock),
                     'KANDEV_AGENT_DOCKER_TOKEN':self.request['token'],
                     'KANDEV_TASK_DOCKER_ACCESS':'restricted','COMPOSE_PROJECT_NAME':self.project}
                result=subprocess.run(['/usr/local/bin/docker',*self.request['args']],cwd=self.root,
                    env=env,capture_output=True,text=True,timeout=15)
                self.assertEqual(result.returncode,0,result.stderr)
                self.assertEqual(len(self.replays),1)
                before=len(self.renders);env['KANDEV_AGENT_DOCKER_TOKEN']='wrong'
                result=subprocess.run(['/usr/local/bin/docker',*self.request['args']],cwd=self.root,
                    env=env,capture_output=True,text=True,timeout=15)
                self.assertEqual(result.returncode,78)
                self.assertEqual(len(self.renders),before)
            finally:server.shutdown();thread.join(timeout=5)
