class RuntimeModelMixin:
    def command(self, argv, cwd, **kwargs):
        if argv[0] == broker.DOCKER_BIN and argv[1:4] == ['inspect','--type','container']:
            return 1, '', 'No such container in private fixture'
        return super().command(argv, cwd, **kwargs)

    def runtime(self, image=None, invoke=None):
        self.overlay={'services':{'app':{'image':image or self.tag,'read_only':True,
                                       'network_mode':'none','user':'1000:999'}}}
        self.overlay_file.write_text(json.dumps(self.overlay))
        original=self.command
        def config_only(argv,cwd,**kwargs):
            if argv[0]==broker.DOCKER_BIN and argv[-3:]==['config','--format','json']:
                return 0,Path(argv[argv.index('-f')+1]).read_text(),''
            return original(argv,cwd,**kwargs)
        with mock.patch.object(broker,'run_command',side_effect=config_only):
            args=[*self.request['args'][:-4],'config','--format','json']
            result=self.execute(args=args) if invoke is None else invoke(args)
        self.assertEqual(result['returncode'],0)
        self.assertNotIn('build',broker.load_scoped_model(self.root,self.root,self.project)['services']['app'])

    def receipt(self):
        return next((broker.MODEL_DIR).glob('*/successful-builds/*/*.json'))


class BaselineCharacterization(RuntimeModelMixin,fixture.FreshBuildTests):
    # Only this behavioral characterization runs against unchanged installed code.
    def test_actual_current_model_replacement_loses_build_admission(self):
        self.assertFalse(hasattr(broker,'successful_build_images'))
        self.execute()
        self.assertEqual(self.inspect(self.tag)['returncode'],0)
        self.assertEqual(self.inspect(NEW_ID)['returncode'],0)
        self.runtime()
        before=len(self.image_queries)
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):
            self.inspect(self.tag)
        self.assertEqual(len(self.image_queries),before)
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):
            self.inspect(NEW_ID)
        self.assertEqual(len(self.image_queries),before)

    def test_native_history_is_not_an_installed_guarded_route(self):
        for args in (['buildx','history','ls'],
                     ['buildx','history','inspect','default/default/synthetic-ref']):
            with self.assertRaisesRegex(broker.BrokerError,'unsupported task Docker command'):
                self.execute(args=args)
        self.assertEqual(self.renders,[])
        self.assertEqual(self.replays,[])
        self.assertEqual(self.image_queries,[])

    def test_real_wrapper_rejects_native_history_without_engine_call(self):
        endpoint=self.base/'history-private.sock'
        with broker.ThreadedUnixServer(str(endpoint),broker.BrokerHandler) as server:
            server.broker_key=self.key
            thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
            try:
                env={'PATH':'/usr/local/bin:/usr/bin:/bin','HOME':str(self.base),
                     'KANDEV_AGENT_GUARD_ACTIVE':'1','KANDEV_AGENT_GUARD_SCOPE':'project',
                     'KANDEV_TASK_DOCKER_ACCESS':'restricted','COMPOSE_PROJECT_NAME':self.project,
                     'KANDEV_AGENT_DOCKER_ROOT':str(self.root),'KANDEV_AGENT_DOCKER_SOCKET':str(endpoint),
                     'KANDEV_AGENT_DOCKER_TOKEN':self.request['token']}
                result=subprocess.run(['/usr/local/bin/docker','buildx','history','ls'],
                    cwd=self.root,env=env,input='',capture_output=True,text=True,timeout=15)
                self.assertEqual(result.returncode,78,result.stderr)
                self.assertIn('unsupported task Docker command',result.stderr)
                self.assertEqual(self.renders+self.replays+self.image_queries,[])
            finally:server.shutdown();thread.join(timeout=5);self.assertFalse(thread.is_alive())
class ReaderCompatibility(RuntimeModelMixin,fixture.FreshBuildTests):
    def capture_fixture_success(self):
        # A private platform-side capture after the synthetic engine build's
        # zero exit. This is not a live legacy-receipt import or agent API.
        fixture.operations['record_successful_build_output'](
            SimpleNamespace(**vars(broker)),self.root,self.root,self.project,'app',self.tag)

    def test_old_broker_new_reader_retains_proven_output(self):
        self.assertFalse(hasattr(broker,'successful_build_images'))
        self.execute();self.capture_fixture_success();self.runtime()
        endpoint=self.base/'reader-private.sock'
        with broker.ThreadedUnixServer(str(endpoint),broker.BrokerHandler) as server:
            server.broker_key=self.key
            thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()
            try:
                env={'PATH':'/usr/local/bin:/usr/bin:/bin','HOME':str(self.base),
                     'KANDEV_AGENT_GUARD_ACTIVE':'1','KANDEV_AGENT_GUARD_SCOPE':'project',
                     'KANDEV_TASK_DOCKER_ACCESS':'restricted','COMPOSE_PROJECT_NAME':self.project,
                     'KANDEV_AGENT_DOCKER_ROOT':str(self.root),'KANDEV_AGENT_DOCKER_SOCKET':str(endpoint),
                     'KANDEV_AGENT_DOCKER_TOKEN':self.request['token']}
                for target in (self.tag,NEW_ID):
                    result=subprocess.run(['/usr/local/bin/docker','image','inspect','--format','{{.Id}}',target],
                        cwd=self.root,env=env,input='',capture_output=True,text=True,timeout=15)
                    self.assertEqual(result.returncode,0,result.stderr)
                    self.assertEqual(result.stdout.strip(),NEW_ID)
                env['KANDEV_AGENT_DOCKER_TOKEN']='wrong';before=len(self.image_queries)
                result=subprocess.run(['/usr/local/bin/docker','image','inspect',self.tag],
                    cwd=self.root,env=env,input='',capture_output=True,text=True,timeout=15)
                self.assertEqual(result.returncode,78)
                self.assertEqual(len(self.image_queries),before)
            finally:server.shutdown();thread.join(timeout=5);self.assertFalse(thread.is_alive())

    def test_old_broker_without_success_receipt_stays_denied(self):
        self.execute();self.runtime();self.image_queries.clear()
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):self.inspect(self.tag)
        self.assertEqual(self.image_queries,[])

    def test_old_broker_reader_rejects_changed_output(self):
        self.execute();self.capture_fixture_success();self.runtime();self.image_queries.clear()
        original=self.command
        def retarget(argv,cwd,**kwargs):
            if argv[1:3]==['image','inspect'] and argv[-1]==self.tag:
                self.image_queries.append(argv);return 0,json.dumps(OLD_ID),''
            return original(argv,cwd,**kwargs)
        with mock.patch.object(broker,'run_command',side_effect=retarget):
            with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):self.inspect(self.tag)
        self.assertTrue(all(call[-1]==self.tag for call in self.image_queries))


class HistoryTests(RuntimeModelMixin,fixture.FreshBuildTests):
    def test_successful_output_survives_runtime_replacement(self):
        self.execute();self.runtime()
        self.assertEqual(self.inspect(self.tag)['returncode'],0)
        self.assertEqual(self.inspect(NEW_ID)['returncode'],0)
        record=broker.successful_build_images(self.root,self.root)[self.tag]
        self.assertEqual(record['image_id'],NEW_ID)
        self.assertEqual(record['kind'],'successful-build-output')
        self.assertEqual(self.inspect(self.tag)['stdout'],self.inspect(NEW_ID)['stdout'])

    def test_model_only_runtime_reference_creates_no_history(self):
        self.runtime()
        self.assertEqual(broker.successful_build_images(self.root,self.root),{})
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):self.inspect(self.tag)
        self.assertEqual(self.image_queries,[])

    def test_failed_build_does_not_renew_or_create_history(self):
        self.engine_exit=19
        self.execute();self.runtime()
        self.assertEqual(broker.successful_build_images(self.root,self.root),{})
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):self.inspect(self.tag)
        self.assertEqual(self.image_queries,[])

    def test_config_with_build_is_not_a_success_receipt(self):
        self.test_config_registers_model_without_build_or_output_proof()
        self.runtime()
        self.assertEqual(broker.successful_build_images(self.root,self.root),{})
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):self.inspect(self.tag)

    def test_historical_tag_retargeting_cannot_admit_changed_image(self):
        self.execute();self.runtime();self.image_queries.clear()
        original=self.command
        def retarget(argv,cwd,**kwargs):
            if argv[1:3]==['image','inspect'] and argv[-1]==self.tag:
                self.image_queries.append(argv);return 0,json.dumps(OLD_ID),''
            return original(argv,cwd,**kwargs)
        with mock.patch.object(broker,'run_command',side_effect=retarget):
            for target in (self.tag,NEW_ID,OLD_ID):
                with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):self.inspect(target)
        self.assertTrue(all(call[-1]==self.tag for call in self.image_queries))

    def test_sibling_and_other_task_do_not_inherit_receipts(self):
        self.execute();self.runtime();self.image_queries.clear()
        sibling=self.root/'sibling';sibling.mkdir()
        for root,cwd in [(self.root,sibling),(self.base,self.base)]:
            self.assertEqual(broker.successful_build_images(root,cwd),{})
            request={**self.request,'task_root':str(root),'cwd':str(cwd),
                     'token':broker.expected_token(self.key,root),'args':['image','inspect',self.tag]}
            with self.assertRaisesRegex(broker.BrokerError,'IMAGE_NOT_AUTHORIZED'):
                broker.handle_request(request,self.key)
        self.assertEqual(self.image_queries,[])

    def test_receipt_custody_and_binding_fail_closed(self):
        self.execute();self.runtime();self.image_queries.clear()
        path=self.receipt();original=path.read_bytes()
        record=json.loads(original)
        for changes in ({'cwd':str(self.base)},{'task_root':str(self.base)},
                        {'image_id':'not-an-id'},{'image':'global:tag'}):
            path.write_text(json.dumps({**record,**changes}))
            with self.assertRaisesRegex(broker.BrokerError,'IMAGE_METADATA_UNAVAILABLE'):self.inspect(self.tag)
        path.write_bytes(original);path.chmod(0o666)
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_METADATA_UNAVAILABLE'):self.inspect(self.tag)
        path.chmod(0o600);destination=path.with_suffix('.private');path.rename(destination);path.symlink_to(destination)
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_METADATA_UNAVAILABLE'):self.inspect(self.tag)
        self.assertEqual(self.image_queries,[])

    def test_claim_revocation_blocks_retained_output(self):
        self.execute();self.runtime();self.image_queries.clear()
        broker.project_claim_path(self.project).unlink()
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_METADATA_UNAVAILABLE'):self.inspect(self.tag)
        self.assertEqual(self.image_queries,[])

    def test_history_metadata_remains_curated_and_typed(self):
        self.execute();self.runtime();self.image_queries.clear()
        with self.assertRaisesRegex(broker.BrokerError,'IMAGE_METADATA_FORMAT'):
            broker.handle_request({**self.request,'args':['image','inspect','--format','{{json .Config.Env}}',self.tag]},self.key)
        with self.assertRaises(broker.BrokerError):
            broker.handle_request({**self.request,'args':['container','inspect',self.tag]},self.key)
        self.assertEqual(self.image_queries,[])
        result=broker.handle_request({**self.request,'args':['image','inspect','--format','{{.Id}}',self.tag]},self.key)
        self.assertEqual(result['stdout'].strip(),NEW_ID)

    def test_dry_run_print_check_and_ambiguous_modes_cannot_attest(self):
        for args in (['--print','app'],['--check','app'],['--push','app'],
                     ['--build-arg','app'],['app','db'],[],['--with-dependencies','app']):
            self.assertIsNone(broker.selected_receipt_service(self.merged(),args))
        args=self.request['args'];self.execute(args=[args[0],'--dry-run',*args[1:]])
        self.runtime()
        self.assertEqual(broker.successful_build_images(self.root,self.root),{})

        with mock.patch.object(broker,'record_successful_build_output',side_effect=OSError('synthetic write failure')):
            result=self.execute()

    def test_standalone_build_records_same_scoped_success_history(self):
        original=self.command
